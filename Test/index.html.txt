  <script>
    function switchTab(tabName) {
      document.querySelectorAll('.nav-item').forEach(el => el.classList.remove('active'));
      event.currentTarget.classList.add('active');

      document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
      document.getElementById('tab-' + tabName).classList.remove('hidden');
    }

    function sendQuickPrompt(promptText) {
      document.getElementById('landing-prompts').classList.add('hidden');
      addMessage(promptText, 'user');
      
      setTimeout(() => {
        let reply = "Decide Shikane is a Full Stack Developer with 4 years of hands-on experience specializing in C#, .NET, React, and SQL Server within fintech and insurance software engineering.";
        if(promptText.includes('technologies')) {
          reply = "Decide specializes in C#, .NET Core / .NET 8, ASP.NET Web API, React, TypeScript, SQL Server, RabbitMQ, REST APIs, and Docker.";
        } else if(promptText.includes('projects')) {
          reply = "Decide built the Customer Spending Insights Dashboard (React/TypeScript/Docker) and RevenuePilot AI (.NET 8/PostgreSQL churn prediction engine).";
        } else if(promptText.includes('education')) {
          reply = "Decide holds a National Diploma in Information Technology (Software Development) from Tshwane University of Technology.";
        }
        addMessage(reply, 'bot');
      }, 500);
    }

    function sendMessage() {
      const input = document.getElementById('user-input');
      const text = input.value.trim();
      if(!text) return;

      document.getElementById('landing-prompts').classList.add('hidden');
      addMessage(text, 'user');
      input.value = '';

      setTimeout(() => {
        addMessage("Decide is an experienced Full Stack Software Developer with core strengths in C#/.NET backend development, React frontends, and SQL database performance.", 'bot');
      }, 600);
    }

    function addMessage(text, sender) {
      const history = document.getElementById('chat-history');
      const msgDiv = document.createElement('div');
      msgDiv.className = `chat-message ${sender}`;
      msgDiv.innerHTML = `<div class="msg-body">${text}</div>`;
      history.appendChild(msgDiv);
      history.scrollTop = history.scrollHeight;
    }
  </script>
